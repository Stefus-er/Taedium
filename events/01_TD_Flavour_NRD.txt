country_event = { #The Mercenaries of the North
	id = nordmark.1
	title = nordmark.1.t
	desc = nordmark.1.desc
	picture = DISCIPLINED_SOLDIERS_eventPicture

	fire_only_once = yes

	trigger = {
		current_age = age_of_discovery
		OR = {
			primary_culture = nahtrucer
			primary_culture = nordmarker
		}
		OR = {
			EHR = {
				war_with = RAN
			}
			KRI = {
				war_with = BOG
			}
		}
		OR = {	#Nahtrucer country is renting condottieri to winning side.
			AND = {
				is_renting_condottieri_to = EHR
				EHR = {
					war_score = 10
				}
			}
			AND = {
				is_renting_condottieri_to = RAN
				RAN = {
					war_score = 10
				}
			}
			AND = {
				is_renting_condottieri_to = KRI
				KRI = {
					war_score = 10
				}
			}
			AND = {
				is_renting_condottieri_to = BOG
				BOG = {
					war_score = 10
				}
			}
		}
	}

	mean_time_to_happen = {
		months = 1
	}

	option = {
		name = nordmark.1.a
		add_country_modifier = {
			name = TD_Mercenaries_of_the_North
			duration = -1
			desc = until_end_of_age_desc
		}
		add_army_professionalism = 0.1
	}

	option = {
		name = nordmark.1.b
		add_army_professionalism = 0.05
		add_army_tradition = 20
	}
}

				chance = 25
				add_base_production = 1
			}
			random = {
				chance = 25
				add_base_tax = 1
			}

				chance = 25
				add_base_manpower = 1

country_event = { #Northward Colonisation of Gyrdia
	id = nordmark.10
	title = nordmark.10.t
	desc = nordmark.10.desc
	picture = RELIGIOUS_TURMOIL_eventPicture

	is_triggered_only = yes

	trigger = {
		OR = {
			primary_culture = nahtrucer
			primary_culture = nordmarker
		}
		any_owned_province = {
			religion = wirosepan
		}
		OR = {
			colony = 1
			any_owned_province = {
				OR = {
					continent = chantlia
					continent = venatia
					continent = tubaria
					is_colony = yes
				}
			}
		}
	}

	option = {
		name = nordmark.10.a
		random_owned_province = {
			limit = {
				religion = wirosepan
				continent = mysentria
			}
			random = {
				chance = 33
				change_religion = ROOT
			}
		}
		if = {
			limit = {
				colony = 1
			}
			random_country = {
				limit = {
					is_colonial_nation_of = ROOT
				}
				random_owned_province = {
					limit = {
						OR = {
							continent = chantlia
							continent = venatia
							continent = tubaria
						}
						is_colony = yes
					}
					change_religion = wirosepan
					if = {
						limit = {
							is_colony = yes
						}
						add_colonysize = 200
					}
				}
			}
		}
		else = {
			random_owned_province = {
				limit = {
					OR = {
						continent = chantlia
						continent = venatia
						continent = tubaria
					}
