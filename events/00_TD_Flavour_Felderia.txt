# Foundation of Felderstadt
country_event = {
	id = felderia.2
	title = felderia.2.t
	desc = felderia.1.desc

	picture = DIPLOMACY_SIGNING_eventPicture
	
	is_triggered_only = yes
	trigger = {
		any_owned_province = {
			is_core = ROOT
			has_province_flag = is_felderstadt_flag
		}
	}

	option = { #Glass
		name = felderia.2.a
		random_owned_province = {
			limit = { has_province_flag = is_felderstadt_flag }
			
			change_trade_goods = glass

#Wigebert von Preden
country_event = {
	id = felderia.4
	title = felderia.4.t
	desc = felderia.4.desc

	picture = RELIGIOUS_TURMOIL_eventPicture
	
	mean_time_to_happen = {
		months = 120
		modifier = {
			factor = 0.9
			ai = no #Fuck AI, I want my Player to see the flavour
		}
	}
	fire_only_once = yes
	trigger = {
		primary_culture = felder
		current_age = age_of_reformation
	}

	option = { #State-wide
		name = felderia.4.a
		if = {
			limit = {
				religion = sigilian
			}
			add_ruler_modifier = {
				name = wigebert_sigillian_mod
			}
		}
		else = {
			add_ruler_modifier = {
				name = wigebert_non_sigillian_mod
			}
		}
	}
	option = { #Advisor
		name = felderia.4.b
		define_advisor = {
			type = theologian
			name = "Wigebert von Preden"
			culture = felder
			skill = 3
			cost_multiplier = 0.25
		}
	}
}

#Eben Harvest
country_event = {
	id = felderia.7
	title = felderia.7.t
	desc = felderia.7.desc

	picture = TRIBUTARY_FLEET_eventPicture
	
	is_triggered_only = yes
	trigger = {
		OR = {
			primary_culture = felder
			capital_scope = {
				province_group = eben_riverbed_group
			}
		}
		NOT = {
			any_owned_province = {
				has_province_modifier = eben_harvest_mod
			}
		}
		OR = {
			NOT = { has_country_flag = had_eben_harvest_flag }
			had_country_flag = { flag = had_eben_harvest_flag days = 3650 }
		}
	}
	
	immediate = {
		hidden_effect = {
			clr_country_flag = had_eben_harvest_flag
		}
	}

	after = {
		hidden_effect = {
			set_country_flag = had_eben_harvest_flag
		}
	}

	option = { #Excellent!
		name = felderia.7.a
		add_years_of_owned_provinces_production_income = {
			years = 0.5
		}
		random_owned_province = {
			limit = {
				culture = felder
				OR = {
					trade_goods = wine
					trade_goods = grain
					trade_goods = livestock
				}
			}
			add_province_modifier = {
				name = eben_harvest_mod
				#desc = eben_harvest_mod_desc
				duration = 1095
			}
		}
	}
}
