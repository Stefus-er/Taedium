		}
	}
}

country_event = { #Die Gustenwein (The Gusten Wine)

	fire_only_once = yes

	trigger = {
		prestige = 50
		NOT = { current_age = age_of_discovery }
		OR = {
			owns = 570
			owns = 603
		}
	}

	mean_time_to_happen = {
		months = 1
	}

	option = {
		name = salia.2.a
		add_splendor = 100
		if = {
			limit = {
				current_age = age_of_reformation
			}
			add_country_modifier = {
				name = TD_Die_Gustenwein
				duration = -1
				desc = until_end_of_age_desc
			}
		}
		else = {
			add_prestige = 25
		}
	}
}
