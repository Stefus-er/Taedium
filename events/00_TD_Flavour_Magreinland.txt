			}
			else = {
				add_base_tax = 2
			}
			if = {
				limit = {
					NOT = {
						has_building = courthouse
					}
				}
				add_building = courthouse
			}
			else = {
				add_base_tax = 2
			}
		}
	}

	id = magreiner.4
	title = magreiner.4.t
	desc = magreiner.4.desc

	mean_time_to_happen = { #Should happen between 15 to 30 years, forcing it into bi_yearly_pulse would be kinda counter-intuitive
		months = 360
	}
	trigger = {
		capital_scope = {
			region = magreinland_region
		}
		OR = {
			NOT = { has_global_flag = magreina_flood_flag } #This Trigger Exists solely as a minimum cap for this even to happen, beacuse without it if player is unlucky enough the flood could happen very, VERY frequently
			had_global_flag = { flag = magreina_flood_flag days = 3650 }
		}
		NOT = {
			643 = {
				has_great_project = {
					type = magrein_dykes
					tier = 1
				}
			}
		}
	}
	major = yes
	major_trigger = {
		OR = {
			capital_scope = {
				region = magreinland_region

		hidden_effect = {
			clr_global_flag = magreina_flood_flag

#Ludwig von Ertzberg
country_event = {
	id = magreiner.5
	title = magreiner.5.t
	desc = magreiner.5.desc

	picture = PAINTER_ARTIST_eventPicture
	
	mean_time_to_happen = {
		months = 300
		modifier = {
			factor = 0.5
			OR = {
				government_rank = 2
				total_development = 300
			}
		}
	}
	fire_only_once = yes
	trigger = {
		OR = {
			primary_culture = upper_magreiner
			primary_culture = middle_magreiner
		}
		is_year = 1550
		NOT = {
			is_year = 1650
		}
	}

	option = { #Advisor
		name = magreiner.5.a
		define_advisor = {
			type = artist
			name = "Ludwig von Ertzberg"
			culture = upper_magreiner
			skill = 3
			cost_multiplier = 0.5
		}
		add_splendor = 100
		add_prestige = 25
	}
}

#Federalisation of Magreinland
country_event = {
	id = magreiner.9
	title = magreiner.9.t
	desc = magreiner.9.desc

	picture = REFORM_eventPicture
	
	is_triggered_only = yes

	option = {
		name = magreiner.9.a
		unlock_government_reform = { government_reform = magreiner_federal_monarchy_reform }
		if = {
			limit = {
				NOT = {
					government = monarchy #I know this wasn't in the draft, but the chances are someone swaps government to monarchy are low as fuck, and without it Rewards seems lackluster.
				}
			}
			add_adm_power = 100
			add_dip_power = 100
			add_prestige_or_monarch_power = {
				amount = 50
			}
		}
		add_innovativeness_or_monarch_power = { amount = 25 }
		custom_tooltip = kill_unrest_separatism_magreiner_tt
		hidden_effect = {
			every_owned_province = {
				limit = {
					OR = {
						culture = upper_magreiner
						culture = middle_magreiner
					}
				}
				add_unrest = -1000
				add_nationalism = -1000
			}
		}
	}
}
