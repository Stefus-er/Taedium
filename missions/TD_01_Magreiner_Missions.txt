TD_Branch_Toflanter = {
	slot = 4
	generic = no
	ai = yes
	has_country_shield = yes
	potential = { 
		primary_culture = middle_magreiner 
		NOT = {
			tag = KRI
		}
	} 

	TD_Toflanter_Peasantry = {
		icon = mission_peasantry
		position = 2
		provinces_to_highlight = {  } 

		trigger = {
			NOT = {
				has_estate_privilege = estate_peasants_imperial_edict
			}
			OR = {
				AND = {
					estate_loyalty = {
						estate = estate_nobles
						loyalty = 60
					}
					has_estate_privilege = estate_nobles_protect_rural_nobility
				}
				AND = {
					estate_loyalty = {
						estate = estate_peasants
						loyalty = 60
					}
					has_estate_privilege = estate_peasants_expanded_freedoms
				}
			}
		} 

		effect = {
			custom_tooltip = TD_Toflanter_Empower_Nobles_tt
			if = {
				limit = {
					estate_loyalty = {
						estate = estate_nobles
						loyalty = 60
					}
					has_estate_privilege = estate_nobles_protect_rural_nobility
				}
				set_country_flag = toflanter_upgrade_rural_nobility
			}
			custom_tooltip = TD_Toflanter_Empower_Peasants_tt
			if = {
				limit = {
					estate_loyalty = {
						estate = estate_peasants
						loyalty = 60
					}
					has_estate_privilege = estate_peasants_expanded_freedoms
				}
				set_country_flag = toflanter_upgrade_peasant_freedoms
			}
		} 
	}
}

		}
	} 

	TD_Ordenic_Estates = {
		icon = balance_the_estates
		position = 2
		provinces_to_highlight = {  } 

		trigger = {
			custom_trigger_tooltip = {
				tooltip = atleast_three_estates_loyal_tt
				calc_true_if = {
					estate_loyalty = {
						estate = estate_nobles
						loyalty = 60
					}
					estate_loyalty = {
						estate = estate_burghers
						loyalty = 60
					}
					estate_loyalty = {
						estate = estate_church
						loyalty = 60
					}
					estate_loyalty = {
						estate = estate_peasants
						loyalty = 60
					}
					amount = 3
				}
			}
			num_of_estate_agendas_completed = {
				estate = estate_nobles
				value = 1
			}
			num_of_estate_agendas_completed = {
				estate = estate_burghers
				value = 1
			}
			num_of_estate_agendas_completed = {
				estate = estate_church
				value = 1
			}
			num_of_estate_agendas_completed = {
				estate = estate_peasants
				value = 1
			}
		} 

		effect = {
			add_stability_or_adm_power = 1
			if = {
				limit = {
					has_estate = estate_burghers
					estate_loyalty = {
						estate = estate_burghers
						loyalty = 60
					}
				}
				add_estate_influence_modifier = {
					estate = estate_burghers
					influence = -0.1
					duration = 9125
				}
			}
			if = {
				limit = {
					has_estate = estate_church
					estate_loyalty = {
						estate = estate_church
						loyalty = 60
					}
				}
				add_estate_influence_modifier = {
					estate = estate_church
					influence = -0.1
					duration = 9125
				}
			}
			if = {
				limit = {
					has_estate = estate_nobles
					estate_loyalty = {
						estate = estate_nobles
						loyalty = 60
					}
				}
				add_estate_influence_modifier = {
					estate = estate_nobles
					influence = -0.1
					duration = 9125
				}
			}
			if = {
				limit = {
					has_estate = estate_peasants
					estate_loyalty = {
						estate = estate_peasants
						loyalty = 60
					}
				}
				add_estate_influence_modifier = {
					estate = estate_peasants
					influence = -0.1
					duration = 9125
				}
			}
			add_country_modifier = {
				name = TD_Unity_of_Estates
				duration = 9125
			}
			custom_tooltip = upgraded_ordenic_supremacy_over_crown_tt
			set_country_flag = upgraded_ordenic_supremacy_over_crown_flag
		} 
	}
}

				}
			}
			hidden_effect = {
				every_owned_province = {
					limit = {
						OR = {

							province_id = 642
							province_id = 495
							province_id = 544
							province_id = 539
							province_id = 685
						}
					}
					if = {
						limit = {
							NOT = {

							}

					}
					else = {
						add_base_tax = 2
					}
