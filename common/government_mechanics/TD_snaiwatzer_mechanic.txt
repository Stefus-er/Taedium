snaiwatzer_army_loyalty_mechanic = {
	alert_icon_gfx = GFX_alerticons_government_mechanics
	alert_icon_index = 11
	available = {
		
	}

	powers = {
		army_loyalty_power = {
			#gui = snaiwatzer_loyalty_mechanic_gui
			min = 0
			max = 100
			default = 100
			is_good = yes
			scaled_modifier = {
				modifier = {
					land_maintenance_modifier = -0.1
					possible_condottieri = 0.5
				}
		        start_value = 0
		        end_value = 100
		        #extend_beyond_value = no
			}
			on_min_reached = {
				log = "Elenders: army loyalty 0"
				if = {
					limit = { #Elender event chain has already started
						OR = {
							has_global_flag = ELENDERS_BORN
							any_known_country = { has_country_flag = ELENDERS_SPAWN_TARGET }
						}
					}
				} else = { #start the chain
					set_global_flag = elenders_activation_flag
					random_owned_province = {
						limit = { is_capital = no culture = snaiwatzer }
						add_province_triggered_modifier = elenders_activation
					}
					set_country_flag = ELENDERS_SPAWN_TARGET
				}
			}
		}
	}
}
